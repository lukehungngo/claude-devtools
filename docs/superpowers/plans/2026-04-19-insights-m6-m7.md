```typescript
import { openSync, readSync, statSync, closeSync } from "fs";
import { calculateTokenCost } from "./metrics.js";
import { getTimeRangeCutoff } from "./insights-aggregator.js";
import type {
  SessionInfo,
  InsightsBreakdown,
  InsightsTimeRange,
} from "../types.js";

interface BreakdownSessionData {
  fileSize: number;
  mtime: number;
  models: Map<string, { tokensIn: number; tokensOut: number; cost: number; turns: number }>;
  tools: Map<string, number>;
  totalCost: number;
  totalTokens: number;
}

const breakdownCache = new Map<string, BreakdownSessionData>();

function readAllLines(filePath: string): unknown[] {
  const fd = openSync(filePath, "r");
  try {
    const stat = statSync(filePath);
    if (stat.size === 0) return [];
    const buf = Buffer.allocUnsafe(stat.size);
    readSync(fd, buf, 0, stat.size, 0);
    return buf
      .toString("utf8")
      .split("\n")
      .filter((l) => l.trim())
      .flatMap((line) => {
        try {
          return [JSON.parse(line)];
        } catch {
          return [];
        }
      });
  } finally {
    closeSync(fd);
  }
}

function computeBreakdownSessionData(session: SessionInfo): BreakdownSessionData {
  const stat = statSync(session.path);
  const cached = breakdownCache.get(session.path);
  if (cached && cached.fileSize === stat.size && cached.mtime === stat.mtimeMs) {
    return cached;
  }

  const models = new Map<string, { tokensIn: number; tokensOut: number; cost: number; turns: number }>();
  const tools = new Map<string, number>();
  let totalCost = 0;
  let totalTokens = 0;

  for (const raw of readAllLines(session.path)) {
    const event = raw as {
      type?: string;
      message?: {
        model?: string;
        usage?: {
          input_tokens?: number;
          output_tokens?: number;
          cache_creation_input_tokens?: number;
          cache_read_input_tokens?: number;
        };
        content?: Array<{ type: string; name?: string }>;
      };
    };

    if (event.type !== "assistant" || !event.message) continue;
    const msg = event.message;
    if (!msg.model || !msg.usage) continue;

    const model = msg.model;
    const inTok = msg.usage.input_tokens ?? 0;
    const outTok = msg.usage.output_tokens ?? 0;
    const cost = calculateTokenCost(model, {
      inputTokens: inTok,
      outputTokens: outTok,
      cacheWriteTokens: msg.usage.cache_creation_input_tokens ?? 0,
      cacheReadTokens: msg.usage.cache_read_input_tokens ?? 0,
    });

    const existing = models.get(model) ?? { tokensIn: 0, tokensOut: 0, cost: 0, turns: 0 };
    models.set(model, {
      tokensIn: existing.tokensIn + inTok,
      tokensOut: existing.tokensOut + outTok,
      cost: existing.cost + cost,
      turns: existing.turns + 1,
    });
    totalCost += cost;
    totalTokens += inTok + outTok;

    if (Array.isArray(msg.content)) {
      for (const item of msg.content) {
        if (item.type === "tool_use" && item.name) {
          tools.set(item.name, (tools.get(item.name) ?? 0) + 1);
        }
      }
    }
  }

  const result: BreakdownSessionData = {
    fileSize: stat.size,
    mtime: stat.mtimeMs,
    models,
    tools,
    totalCost,
    totalTokens,
  };
  breakdownCache.set(session.path, result);
  return result;
}

export function computeInsightsBreakdown(
  sessions: SessionInfo[],
  timeRange: InsightsTimeRange,
  repo: string
): InsightsBreakdown {
  const cutoff = getTimeRangeCutoff(timeRange);

  const filtered = sessions.filter((s) => {
    const t = new Date(s.startTime).getTime();
    if (t < cutoff) return false;
    if (repo !== "all" && s.cwd !== repo && !s.cwd?.includes(repo)) return false;
    return true;
  });

  const modelMap = new Map<string, { tokensIn: number; tokensOut: number; cost: number; turns: number }>();
  const toolMap = new Map<string, number>();
  const repoMap = new Map<string, { tokens: number; cost: number }>();
  const sessionCosts: Array<{ id: string; label: string; cost: number }> = [];

  for (const session of filtered) {
    const data = computeBreakdownSessionData(session);

    for (const [model, stats] of data.models) {
      const existing = modelMap.get(model) ?? { tokensIn: 0, tokensOut: 0, cost: 0, turns: 0 };
      modelMap.set(model, {
        tokensIn: existing.tokensIn + stats.tokensIn,
        tokensOut: existing.tokensOut + stats.tokensOut,
        cost: existing.cost + stats.cost,
        turns: existing.turns + stats.turns,
      });
    }

    for (const [name, count] of data.tools) {
      toolMap.set(name, (toolMap.get(name) ?? 0) + count);
    }

    const slug = session.cwd ?? session.path;
    const repoExisting = repoMap.get(slug) ?? { tokens: 0, cost: 0 };
    repoMap.set(slug, {
      tokens: repoExisting.tokens + data.totalTokens,
      cost: repoExisting.cost + data.totalCost,
    });

    if (data.totalCost > 0) {
      const date = session.startTime.slice(0, 10);
      const repoName = session.cwd ? session.cwd.split("/").pop() ?? "session" : "session";
      sessionCosts.push({
        id: session.id,
        label: `${repoName} · ${date}`,
        cost: data.totalCost,
      });
    }
  }

  const totalCost = [...modelMap.values()].reduce((s, m) => s + m.cost, 0);
  const models = [...modelMap.entries()]
    .map(([model, stats]) => ({
      model,
      ...stats,
      share: totalCost > 0 ? (stats.cost / totalCost) * 100 : 0,
    }))
    .sort((a, b) => b.cost - a.cost);

  const topTools = [...toolMap.entries()]
    .map(([name, calls]) => ({ name, calls }))
    .sort((a, b) => b.calls - a.calls)
    .slice(0, 10);

  const topRepos = [...repoMap.entries()]
    .map(([slug, stats]) => ({ slug, ...stats }))
    .sort((a, b) => b.cost - a.cost)
    .slice(0, 5);

  const topSessions = sessionCosts
    .sort((a, b) => b.cost - a.cost)
    .slice(0, 5);

  return { models, topRepos, topSessions, topTools };
}
```

```typescript
import { openSync, readSync, statSync, closeSync } from "fs";
import { getTimeRangeCutoff } from "./insights-aggregator.js";
import type {
  SessionInfo,
  InsightsTrends,
  InsightsTrendEntry,
  InsightsTimeRange,
  TrendVerdict,
} from "../types.js";

const WEEK_MS = 7 * 24 * 60 * 60 * 1000;

interface TrendSessionData {
  fileSize: number;
  mtime: number;
  commands: Map<string, { tokensIn: number; tokensOut: number }>; // commandName → per-run tokens
  agents: Map<string, { tokensIn: number; tokensOut: number }>;
  skills: Map<string, { tokensIn: number; tokensOut: number }>;
}

const trendsCache = new Map<string, TrendSessionData>();

function readAllLines(filePath: string): unknown[] {
  const fd = openSync(filePath, "r");
  try {
    const stat = statSync(filePath);
    if (stat.size === 0) return [];
    const buf = Buffer.allocUnsafe(stat.size);
    readSync(fd, buf, 0, stat.size, 0);
    return buf
      .toString("utf8")
      .split("\n")
      .filter((l) => l.trim())
      .flatMap((line) => {
        try { return [JSON.parse(line)]; }
        catch { return []; }
      });
  } finally {
    closeSync(fd);
  }
}

function computeTrendSessionData(session: SessionInfo): TrendSessionData {
  const stat = statSync(session.path);
  const cached = trendsCache.get(session.path);
  if (cached && cached.fileSize === stat.size && cached.mtime === stat.mtimeMs) {
    return cached;
  }

  const commands = new Map<string, { tokensIn: number; tokensOut: number }>();
  const agents = new Map<string, { tokensIn: number; tokensOut: number }>();
  const skills = new Map<string, { tokensIn: number; tokensOut: number }>();

  // Track last assistant token context for attribution
  let lastInTok = 0;
  let lastOutTok = 0;

  for (const raw of readAllLines(session.path)) {
    const event = raw as {
      type?: string;
      userType?: string;
      message?: {
        role?: string;
        usage?: { input_tokens?: number; output_tokens?: number };
        content?: Array<{
          type: string;
          text?: string;
          name?: string;
          input?: Record<string, unknown>;
        }>;
      };
    };

    if (event.type === "assistant" && event.message) {
      lastInTok = event.message.usage?.input_tokens ?? 0;
      lastOutTok = event.message.usage?.output_tokens ?? 0;

      // Detect agent + skill invocations
      if (Array.isArray(event.message.content)) {
        for (const item of event.message.content) {
          if (item.type !== "tool_use") continue;

          if (item.name === "Agent") {
            const agentName =
              (item.input?.subagent_type as string) ??
              (item.input?.description as string)?.split(" ")[0] ??
              "unknown";
            const existing = agents.get(agentName) ?? { tokensIn: 0, tokensOut: 0 };
            agents.set(agentName, {
              tokensIn: existing.tokensIn + lastInTok,
              tokensOut: existing.tokensOut + lastOutTok,
            });
          }

          if (item.name === "Skill") {
            const skillName = (item.input?.skill as string) ?? "unknown";
            const existing = skills.get(skillName) ?? { tokensIn: 0, tokensOut: 0 };
            skills.set(skillName, {
              tokensIn: existing.tokensIn + lastInTok,
              tokensOut: existing.tokensOut + lastOutTok,
            });
          }
        }
      }
    }

    if (event.type === "user" && event.userType === "external" && event.message) {
      const content = event.message.content;
      if (!Array.isArray(content)) continue;
      const first = content[0];
      if (first?.type !== "text" || !first.text?.startsWith("/")) continue;

      // Extract command name (first word, e.g. "/review")
      const cmdName = first.text.split(/\s+/)[0];
      const existing = commands.get(cmdName) ?? { tokensIn: 0, tokensOut: 0 };
      commands.set(cmdName, {
        tokensIn: existing.tokensIn + lastInTok,
        tokensOut: existing.tokensOut + lastOutTok,
      });
    }
  }

  const result: TrendSessionData = {
    fileSize: stat.size,
    mtime: stat.mtimeMs,
    commands,
    agents,
    skills,
  };
  trendsCache.set(session.path, result);
  return result;
}

function linearRegressionSlope(y: number[]): number {
  const n = y.length;
  if (n < 2) return 0;
  const sumX = (n * (n - 1)) / 2;
  const sumY = y.reduce((s, v) => s + v, 0);
  const sumXY = y.reduce((s, v, i) => s + i * v, 0);
  const sumX2 = (n * (n - 1) * (2 * n - 1)) / 6;
  const denom = n * sumX2 - sumX * sumX;
  if (denom === 0) return 0;
  return (n * sumXY - sumX * sumY) / denom;
}

function computeVerdict(weeklyTotals: number[]): TrendVerdict {
  if (weeklyTotals.length < 2) return "stable";
  const slope = linearRegressionSlope(weeklyTotals);
  const avg = weeklyTotals.reduce((s, v) => s + v, 0) / weeklyTotals.length;
  if (avg === 0) return "stable";
  const relSlope = slope / avg;
  if (relSlope > 0.05) return "improving";
  if (relSlope < -0.05) return "regressing";
  return "stable";
}

function buildTrendEntries(
  entityWeeklyMap: Map<string, Array<{ tokensIn: number; tokensOut: number; calls: number }>>,
  numWeeks: number
): InsightsTrendEntry[] {
  return [...entityWeeklyMap.entries()]
    .map(([name, weeks]) => {
      // Ensure all week buckets exist
      while (weeks.length < numWeeks) {
        weeks.push({ tokensIn: 0, tokensOut: 0, calls: 0 });
      }
      const totalCalls = weeks.reduce((s, w) => s + w.calls, 0);
      const totalIn = weeks.reduce((s, w) => s + w.tokensIn, 0);
      const totalOut = weeks.reduce((s, w) => s + w.tokensOut, 0);
      const weekly = weeks.map((w) => ({ in: w.tokensIn, out: w.tokensOut }));
      const weeklyTotals = weeks.map((w) => w.tokensIn + w.tokensOut);
      return {
        name,
        calls: totalCalls,
        avgIn: totalCalls > 0 ? Math.round(totalIn / totalCalls) : 0,
        avgOut: totalCalls > 0 ? Math.round(totalOut / totalCalls) : 0,
        weekly,
        verdict: computeVerdict(weeklyTotals),
      };
    })
    .sort((a, b) => b.calls - a.calls)
    .slice(0, 20);
}

export function computeInsightsTrends(
  sessions: SessionInfo[],
  timeRange: InsightsTimeRange,
  repo: string
): InsightsTrends {
  const cutoff = getTimeRangeCutoff(timeRange);
  const now = Date.now();

  // Build weekly buckets
  const totalMs = now - cutoff;
  const numWeeks = Math.max(1, Math.ceil(totalMs / WEEK_MS));
  const bucketStart = now - numWeeks * WEEK_MS;

  const filtered = sessions.filter((s) => {
    const t = new Date(s.startTime).getTime();
    if (t < cutoff) return false;
    if (repo !== "all" && s.cwd !== repo && !s.cwd?.includes(repo)) return false;
    return true;
  });

  // Maps: entityName → array of numWeeks buckets
  const commandWeeks = new Map<string, Array<{ tokensIn: number; tokensOut: number; calls: number }>>();
  const agentWeeks = new Map<string, Array<{ tokensIn: number; tokensOut: number; calls: number }>>();
  const skillWeeks = new Map<string, Array<{ tokensIn: number; tokensOut: number; calls: number }>>();

  function ensureWeeks(map: Map<string, Array<{ tokensIn: number; tokensOut: number; calls: number }>>, name: string) {
    if (!map.has(name)) {
      map.set(name, Array.from({ length: numWeeks }, () => ({ tokensIn: 0, tokensOut: 0, calls: 0 })));
    }
    return map.get(name)!;
  }

  function weekIndex(sessionTime: number): number {
    const offset = sessionTime - bucketStart;
    return Math.min(numWeeks - 1, Math.max(0, Math.floor(offset / WEEK_MS)));
  }

  for (const session of filtered) {
    const data = computeTrendSessionData(session);
    const t = new Date(session.startTime).getTime();
    const wi = weekIndex(t);

    for (const [name, stats] of data.commands) {
      const weeks = ensureWeeks(commandWeeks, name);
      weeks[wi].calls += 1;
      weeks[wi].tokensIn += stats.tokensIn;
      weeks[wi].tokensOut += stats.tokensOut;
    }

    for (const [name, stats] of data.agents) {
      const weeks = ensureWeeks(agentWeeks, name);
      weeks[wi].calls += 1;
      weeks[wi].tokensIn += stats.tokensIn;
      weeks[wi].tokensOut += stats.tokensOut;
    }

    for (const [name, stats] of data.skills) {
      const weeks = ensureWeeks(skillWeeks, name);
      weeks[wi].calls += 1;
      weeks[wi].tokensIn += stats.tokensIn;
      weeks[wi].tokensOut += stats.tokensOut;
    }
  }

  return {
    commands: buildTrendEntries(commandWeeks, numWeeks),
    agents: buildTrendEntries(agentWeeks, numWeeks),
    skills: buildTrendEntries(skillWeeks, numWeeks),
  };
}

  const maxIn = Math.max(...weekly.map((w) => w.in), 1);
  const maxTotal = Math.max(...weekly.map((w) => w.in + w.out), 1);

  function xPos(i: number): number {
    return (i / (n - 1)) * W;
  }
  function yIn(v: number): number {
    return H - (v / maxIn) * (H - 2) - 1;
  }
  function yTotal(v: number): number {
    return H - (v / maxTotal) * (H - 2) - 1;
  }

  const inPoints = weekly.map((w, i) => `${xPos(i)},${yIn(w.in)}`).join(" ");
  const totalPoints = weekly.map((w, i) => `${xPos(i)},${yTotal(w.in + w.out)}`).join(" ");

  return (
    <svg width={W} height={H} viewBox={`0 0 ${W} ${H}`} aria-hidden="true">
      <polyline
        points={inPoints}
        fill="none"
        stroke="var(--dt-teal)"
        strokeWidth="1.5"
        strokeLinecap="round"
        strokeLinejoin="round"
      />
      <polyline
        points={totalPoints}
        fill="none"
        stroke="var(--dt-purple)"
        strokeWidth="1.5"
        strokeLinecap="round"
        strokeLinejoin="round"
        strokeDasharray="3 2"
      />
    </svg>
  );
}

export function TrendRow({ entry }: TrendRowProps): JSX.Element {
  return (
    <div className="flex items-center gap-3 py-2 border-b border-dt-border last:border-0">
      <span className="flex-1 text-xs font-mono text-dt-text0 truncate min-w-0">{entry.name}</span>
      <span
        data-testid="trend-row-calls"
        className="text-xs font-mono text-dt-text2 flex-shrink-0 w-16 text-right"
      >
        {entry.calls} calls
      </span>
      <span className="text-xs font-mono text-dt-text2 flex-shrink-0 w-20 text-right hidden sm:block">
        {formatTokens(entry.avgIn)} avg
      </span>
      <DualSparkline weekly={entry.weekly} />
      <span
        data-testid="trend-row-verdict"
        className={[
          "text-xxs font-mono font-semibold px-1.5 py-0.5 rounded-dt-xs flex-shrink-0",
          VERDICT_STYLES[entry.verdict],
        ].join(" ")}
      >
        {VERDICT_LABELS[entry.verdict]}
      </span>
    </div>
  );
}
```
